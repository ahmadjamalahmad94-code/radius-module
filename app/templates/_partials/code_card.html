{# Header/comment line classification — mirrors the onboarding look so a
   sectioned RouterOS script keeps its purple section headers + dim comments. #}

{%- set l = line.lstrip() -%}
{%- if l.startswith('# ─') or l.startswith('# ═') or l.startswith('# ١') or l.startswith('# ٢') or l.startswith('# ٣') or l.startswith('# ٤') or l.startswith('# ٥') or l.startswith('# ٦') or l.startswith('# ٧') or l.startswith('# ٨') or l.startswith('# ٩') -%}
hcode-ln is-hdr
{%- elif l.startswith('#') or l.startswith('//') or l.startswith(';') -%}
hcode-ln is-cmt
{%- else -%}
hcode-ln
{%- endif -%}
{%- endmacro -%}

{# code_explain: the «شرح الكود» panel — pairs each part of the script with a
   plain-Arabic explanation of WHAT it does and WHY. This is the second half of
   the praised onboarding treatment (the first is the visual code card).

   items = list of dicts. Each item: {title, body}. Optional keys make the row a
   clickable table-of-contents entry tied to the code card:
     • start_line / end_line → clicking the row scrolls+highlights that range in
       the code card whose id == `target` (same hooks as code_outline).
     • sec (0-based section index) → a per-section copy button (#-sec-N).
   body is rendered |safe (template-authored Arabic, may contain ). #}
{% macro code_explain(items, target='cc') -%}


  {%- for it in items %}
  

    {{ loop.index }}
    

      

        {{ it.title }}
        {%- if it.sec is defined %}
        
        {%- endif %}
      

      
{{ it.body|safe }}

    

  

  {%- endfor %}


{%- endmacro %}


{# code_explain_ranged: same as code_explain, but injects the per-row jump
   ranges the route computed for THIS page's code card so every row becomes a
   live table-of-contents entry (scroll + flash) instead of a dead row.

   items  = list of {title, body} dicts (template-authored, may carry `sec`).
   ranges = list aligned with `items`; each entry is a (start_line, end_line)
            pair, or falsy to leave that one row non-interactive (e.g. no card
            on the page). Built in the route from the rendered script so the
            line numbers can never drift from the template.
   The translated text stays here in the template (keeps i18n extraction); only
   the line numbers come from the route. #}
{% macro code_explain_ranged(items, ranges, target='cc') -%}
{%- if ranges -%}{% for it in items %}{% set _r = ranges[loop.index0] if loop.index0 < ranges|length else None %}{% if _r %}{% set _ = it.update({'start_line': _r[0], 'end_line': _r[1]}) %}{% endif %}{% endfor %}{%- endif -%}
{{ code_explain(items, target=target) }}
{%- endmacro %}
